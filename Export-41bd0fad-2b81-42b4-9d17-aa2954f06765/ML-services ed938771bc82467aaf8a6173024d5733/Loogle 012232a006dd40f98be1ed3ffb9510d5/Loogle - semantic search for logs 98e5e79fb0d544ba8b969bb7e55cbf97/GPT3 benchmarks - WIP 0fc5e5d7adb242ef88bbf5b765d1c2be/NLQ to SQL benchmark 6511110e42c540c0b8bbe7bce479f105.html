Questions with direct answers:

  • list all log lines from component COMPONENT_ID that contains an error
SELECT * FROM paiaffluences_logs WHERE componentID = 'COMPONENT_ID' AND logseverity = 'ERROR'
  • list messages that contain connection errors from COMPONENT_X to HOSTNAME_X
SELECT * FROM paiaffluences_logs WHERE message LIKE '%connection error%' AND message LIKE '%COMPONENT_X%' AND message LIKE '%HOSTNAME_X%'
  • where is the first connection error from COMPONENT_X to HOSTNAME_X?
select * from paiaffluences_logs where message like '%connection error%' and message like '%COMPONENT_X%' and message like '%HOSTNAME_X%' order by timestamp asc limit 1
  • list logs with mention of rabbitmq
SELECT * FROM paiaffluences_logs WHERE message LIKE '%rabbitmq%'
  • number of errors with higher score than 30 involving filebeat_topic X:
SELECT count(*) FROM paiaffluences_logs WHERE _score > 30 AND filebeat_topic = 'X’
  • query to list logs from COMPONENT_ID that talk about time-out connections
SELECT * FROM paiaffluences_logs WHERE componentID = 'COMPONENT_ID' AND message LIKE '%timeout%’
  • query to list COMPONENT_ID logs that talk about time-out connections
SELECT * FROM paiaffluences_logs WHERE message LIKE '%timeout%' AND componentID = 'COMPONENT_ID’
  • query to list logs with single occurence
SELECT * FROM paiaffluences_logs WHERE message = '<message>' GROUP BY message HAVING COUNT(*) = 1
  • query to list logs with exactly 10 occurences:
SELECT * FROM paiaffluences_logs WHERE message LIKE '%10%'
  • query to list logs fom COMPONENT_ID_X with less occurences than 20
SELECT * FROM paiaffluences_logs WHERE componentID = 'COMPONENT_ID_X' GROUP BY message HAVING COUNT(*) < 20
  • most occuring ip address in timeout logs:
select count(*) as count, fields_agentID from paiaffluences_logs where message like '%timeout%' group by fields_agentID order by count desc limit 10
  • most occuring ip address in error logs:
select fields_agentID, count() from paiaffluences_logs where logseverity = 'ERROR' group by fields_agentID order by count() desc limit 10
  • most occuring hostname in error logs:
select hostname, count() from paiaffluences_logs where logseverity = 'ERROR' group by hostname order by count() desc limit 10
  • most occuring pod in error logs:
select kubernetes_pod_name, count() from paiaffluences_logs where logseverity = 'ERROR' group by kubernetes_pod_name order by count() desc limit 10
  • query to list most stable pods: 
SELECT kubernetes_pod_name, count(*) as cnt FROM paiaffluences_logs WHERE logseverity = 'INFO' GROUP BY kubernetes_pod_name ORDER BY cnt DESC LIMIT 10
  • container that occurs most in warning logs
SELECT container_id, count(*) as count FROM paiaffluences_logs WHERE logseverity = 'warning' GROUP BY container_id ORDER BY count DESC LIMIT 1
  • most problematic containers:
select container_id, count() from paiaffluences_logs where logseverity = 'ERROR' group by container_id order by count() desc limit 10
  • query to list most problematic hosts:
SELECT hostname, count(*) as count FROM paiaffluences_logs WHERE logseverity = 'ERROR' GROUP BY hostname ORDER BY count DESC
  • how many restarts of pod POD_X?
SELECT count(*) FROM paiaffluences_logs WHERE kubernetes_pod_name = 'POD_X' AND message LIKE '%Restarting container%'
  • pod with most restarts:
select kubernetes_pod_name, count(*) as restarts from paiaffluences_logs where message like '%Restarting container%' group by kubernetes_pod_name order by restarts desc limit 1
  • hostname with most pods
SELECT hostname, count(*) as count FROM paiaffluences_logs GROUP BY hostname ORDER BY count DESC LIMIT 1
  • hostname that contains most restarting pods:
select hostname, count(*) from paiaffluences_logs where message like '%restarting%' group by hostname order by count(*) desc limit 10
  • host with most problematic pods:
select hostname, count(*) as cnt from paiaffluences_logs where logseverity = 'ERROR' group by hostname order by cnt desc limit 10
  • host with most problematic pods yesterday:
SELECT hostname, count(*) as count FROM paiaffluences_logs WHERE logseverity = 'ERROR' AND timestamp > now() - interval '1 day' GROUP BY hostname ORDER BY count DESC LIMIT 1
  • what period of time with most restarts of pods?
SELECT count(*) as count, kubernetes_pod_name, kubernetes_namespace, kubernetes_container_name, kubernetes_node_name, kubernetes_pod_ip, kubernetes_statefulset_name, log_file_path, log_offset, logID, logseverity, message, stream, timestamp, version, _id, _index, _score, _type, componentID, container_id, container_image_name, container_runtime, fields_agentID, fields_clusterName, infraID, filebeat_topic, hostname, input_type
FROM paiaffluences_logs
WHERE message LIKE '%Restarting container%'
GROUP BY kubernetes_pod_name, kubernetes_namespace, kubernetes_container_name, kubernetes_node_name, kubernetes_pod_ip, kubernetes_statefulset_name, log_file_path, log_offset, logID, logseverity, message, stream, timestamp, version, _id, _index, _score, _type, componentID, container_id, container_image_name, container_runtime, fields_agentID, fields_clusterName, infraID, filebeat_topic, hostname, input_type
ORDER BY count DESC
LIMIT 10
  • when was the time of most pod restarts?
select count(*) as count, date_trunc('hour', timestamp) as hour from paiaffluences_logs where message like '%Restarting container%' group by hour order by hour
  • what is time with most restarts of pods of hostnmae HOST?
SELECT hostname, count(*) as count FROM paiaffluences_logs WHERE message LIKE '%restart%' GROUP BY hostname ORDER BY count DESC
