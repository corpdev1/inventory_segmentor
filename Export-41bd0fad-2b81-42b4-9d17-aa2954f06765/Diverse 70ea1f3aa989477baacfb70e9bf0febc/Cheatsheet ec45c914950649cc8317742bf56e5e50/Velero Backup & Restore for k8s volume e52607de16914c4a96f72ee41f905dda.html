Velero Backup & Restore for k8s volume

For velero installation and details, pls refer to this document

create backup

# This command will create a backup immediately
velero schedule create mongodb-scheduled-backup --schedule "@every 2h" \
  --ttl 10h --include-namespaces mongo
# You can fetch the status of the current backup:
velero get backups

create restore

velero create restore --from-backup <mongodb-scheduled-backup_TIMESTAMP>
# for create restore into new namespace 
# https://velero.io/docs/v1.3.0/restore-reference/
velero restore create RESTORE_NAME \
  --from-backup BACKUP_NAME \
  --namespace-mappings old-ns-1:new-ns-1,old-ns-2:new-ns-2

Reduce AZ

velero backup create packetai-backup  --ttl 120h --include-namespaces packetai-system
velero backup create dgraph-backup  --ttl 120h --include-namespaces dgraph
velero backup create mongo-backup  --ttl 120h --include-namespaces mongo
velero backup create od-backup  --ttl 120h --include-namespaces od-system
velero backup create postgres-backup  --ttl 120h --include-namespaces postgres
velero backup create redis-backup  --ttl 120h --include-namespaces redis
velero backup create argo-events-backup  --ttl 120h --include-namespaces argo-events

# take down the ingest
k scale deployment flux --replicas 0 -n flux
k scale deployment --replicas 0 --all -n packetai-system
k scale sts --replicas 0 --all -n packetai-system
k scale deployment --replicas 0 --all -n logstash

# trigger a schedule immediately
# if there is already a schedule, backup will do incremental snapshot
# which is faster than a new backup above
kubectl patch schedule dgraph-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge
kubectl patch schedule mongo-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge
kubectl patch schedule opendistro-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge
kubectl patch schedule postgres-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge
kubectl patch schedule redis-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge
kubectl patch schedule argo-events-backup-schedule  -n velero -p '{"spec":{"schedule":"@every 15m"}}' --type=merge

# set schedule back when a new backup is triggered
