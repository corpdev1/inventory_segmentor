Logstash separating conf file for Kafka and Elasticsearch

The goal is to separate the config of logstash here so that we have one for Elasticsearch and one for Kafka.

This is an example of the config_fb file :

filter {
    if [message] == "" {
        drop {
        }
    }

    if [@metadata][beat] == "filebeat" {
        ruby {
            path => "/etc/logstash/conf_fb/filter_fb.rb"
        }

        if [event][module] == "system" {
            if [fileset][name] == "syslog" {
                grok {
                    match => {
                        "message" => ["%{SYSLOGTIMESTAMP:logdatetime} %{SYSLOGHOST:[system][hostname]} %{DATA:[system][program]}(?:\[%{POSINT:[system][pid]}\])?: %{GREEDYMULTILINE:content}"]
                    }
                    pattern_definitions => {
                        "GREEDYMULTILINE" => "(.|\n)*"
                    }
                }
                mutate {
                    add_field => {
                        "[parsed]" => "true"
                    }
                }

            }
            else if [fileset][name] == "auth" {
                grok {
                    match => {
                        "message" => ["%{SYSLOGTIMESTAMP:logdatetime} %{SYSLOGHOST:[system][hostname]} %{DATA:[system][program]}(?:\[%{POSINT:[system][pid]}\])?: %{GREEDYMULTILINE:content}"]
                    }
                    pattern_definitions => {
                        "GREEDYMULTILINE" => "(.|\n)*"
                    }
                }
                mutate {
                    add_field => {
                        "[parsed]" => "true"
                    }
                }
            }


        }
        else if [event][module] == "apache" {
            grok {
                match => {
                    "message" => [
                    '%{IPORHOST:ip} %{GREEDYDATA} \[%{HTTPDATE:logdatetime}\] \"%{WORD:[apache][access][method]} %{DATA:[apache][access][url]} HTTP/%{NUMBER:[apache][access][http_version]}\" %{NUMBER:[apache][access][response_code]} %{NUMBER:[apache][access][body_sent][bytes]}( \"%{DATA:[apache][access][referrer]}\")?( \"%{DATA:[apache][access][agent]}\")?',
                    'time=%{TIMESTAMP_ISO8601:logdatetime} level=%{WORD:loglevel} msg=\"%{GREEDYDATA:content}\"',
                    "%{IPORHOST:ip} %{GREEDYDATA} \[%{HTTPDATE:logdatetime}\] %{GREEDYDATA:content}",
                    '\[%{HTTPDERROR_DATE:logdatetime}\] \[%{DATA:[apache][error][module]}:%{LOGLEVEL:[apache][error][level]}\] \[pid %{NUMBER:[apache][error][pid]}(:tid %{NUMBER:[apache][error][tid]})?\]( \[client %{HOSTPORT:[apache][error][client]}\])? %{GREEDYDATA:content}',
                    "%{DATA:process_command}: %{GREEDYDATA:content}"
                    ]
                }
            }
            mutate {
                add_field => {
                    "[parsed]" => "true"
                }
            }
            if [apache][access] {
                mutate {
                    add_field => {
                        "content" => "%{[apache][access][method]} HTTP %{[apache][access][http_version]} %{[apache][access][response_code]}  %{[apache][access][response_code]} %{[apache][access][body_sent][bytes]}"
                    }
                }
            }


        }
        else if [event][module] == "mysql" {
            if [fileset][name] == "error" {
                grok {
                    keep_empty_captures => true
                    match => {
                        "message" => ["%{TIMESTAMP_ISO8601:logdatetime} %{NUMBER:[mysql][error][thread_id]} \[%{DATA:[mysql][error][level]}\] %{GREEDYDATA:content}"]
                    }
                }
                mutate {
                    add_field => {
                        "[parsed]" => "true"
                    }
                }
            }

            else if [fileset][name] == 'slowlog' {
                grok {
                    keep_empty_captures => true
                    match => {
                        "message" => ['^# User@Host: %{USER:[mysql][slowlog][user]}(\[[^\]]+\])? @ (%{HOSTNAME:[mysql][slowlog][host]})? \[%{IP:[mysql][slowlog][ip]}\](\s*Id:\s* %{NUMBER:[mysql][slowlog][id]})?\n# Query_time: %{NUMBER:[mysql][slowlog][query_time][sec]}\s* Lock_time: %{NUMBER:[mysql][slowlog][lock_time][sec]}\s* Rows_sent: %{NUMBER:[mysql][slowlog][rows_sent]}\s* Rows_examined: %{NUMBER:[mysql][slowlog][rows_examined]}\n(SET timestamp=%{NUMBER:[mysql][slowlog][timestamp]};\n)?%{GREEDYMULTILINE:[mysql][slowlog][query]}']
                    }
                    pattern_definitions => {
                        "GREEDYMULTILINE" => "(.|\n)*"
                    }
                }

                mutate {
                    add_field => {
                        "[parsed]" => "true"
                    }
                }

            }
        }
        else if [event][module] == "mongodb" {
            grok {
                keep_empty_captures => true
                match => {
                    "message" => [
                    "%{TIMESTAMP_ISO8601:logdatetime} %{MONGO3_SEVERITY:severity} %{MONGO3_COMPONENT:component}%{SPACE}(?:\[%{DATA:context}\])? %{GREEDYDATA:content}",
                    '\{"t":{"\$date":"%{TIMESTAMP_ISO8601:logdatetime}"\},"s":"%{GREEDYDATA}",%{SPACE}"c":"%{GREEDYDATA:c}",%{SPACE}"id":%{DATA},%{SPACE}"ctx":"%{GREEDYDATA}","msg":"%{GREEDYDATA:msg}","attr":%{GREEDYDATA:attr}'
                    ]
                }
            }

            if ("" in [attr]) {
                if "COMMAND" in [c] {

                    grok {
                        keep_empty_captures => true
                        match => {
                            "attr" => '{"type":"%{WORD:command_type}","ns":"%{DATA:ns}","command":{"%{WORD:command_name}":%{GREEDYDATA}'
                        }
                    }
                    mutate {
                        add_field => {
                            "content" => "%{command_name} %{msg} %{command_type} %{ns}"
                        }
                    }
                }
                else {
                    mutate {
                        copy => {
                            "msg" => "content"
                        }

                    }

                }
            }
        }
        else if [event][module] == "nginx" {
            grok {
                keep_empty_captures => true
                match => {
                    "message" => [
                    '\"%{NGINX_DATE:logdatetime} %{DATA:zt}\" client=%{IPORHOST:ip} method=%{GREEDYDATA:content}',
                    '%{DATETIME_YEAR_FIRST:logdatetime} \[%{WORD:loglevel}\] %{NONNEGINT:pid}#%{NONNEGINT:tid}: \*%{NONNEGINT:cid} %{GREEDYDATA:content}'
                    ]
                }
                pattern_definitions => {
                    "NGINX_DATE" => "%{MONTHDAY}/%{MONTH}/%{YEAR}%{TIME}?%{ISO8601_TIMEZONE}?"
                    "DATETIME_YEAR_FIRST" => "%{YEAR}/%{MONTHNUM}/%{MONTHDAY} %{TIME}"
                }
            }
        }
        else if [event][module] == "redis" {
            grok {
                keep_empty_captures => true
                patterns_dir => ["/patterns"]
                match => {
                    "message" => "(%{POSINT:pid}:%{CHAR:role} )?(%{REDISTIMESTAMP1:logdatetime}|%{REDISTIMESTAMP2:logdatetime}) %{REDISLEVEL:loglevel} %{GREEDYDATA:content}"
                }
                pattern_definitions => {
                    "CHAR" => "[a-zA-Z]"
                    "REDISLEVEL" => "[.\-*#]"
                    "REDISTIMESTAMP1" => "%{MONTHDAY} %{MONTH} %{TIME}"
                    "REDISTIMESTAMP2" => "%{MONTHDAY} %{MONTH} %{YEAR} %{TIME}"
                }
            }
        }
        else if [event][module] == "kafka" {
            grok {
                keep_empty_captures => true
                patterns_dir => ["/patterns"]
                match => {
                    "message" => "\[%{DATETIME_YEAR_FIRST:logdatetime}\] %{LOGLEVEL:loglevel} (\[%{DATA:logger}\]%{SPACE})?%{GREEDYDATA:content}"
                }
            }
        }
        else if [event][module] == "rabbitmq" {
            grok {
                keep_empty_captures => true
                patterns_dir => ["/patterns"]
                match => {
                    "message" => [
                    "%{DATETIME_YEAR_FIRST:logdatetime} \[%{LOGLEVEL:loglevel}\] \<%{DATA:process_id}\> %{GREEDYDATA:content}",
                    "%{GREEDYDATA:content}"
                    ]
                }
            }
        }
        else if [event][module] == "postgresql" {
            grok {
                keep_empty_captures => true
                patterns_dir => ["/patterns"]
                match => {
                    "message" => "%{DATETIME_YEAR_FIRST:logdatetime} UTC \[%{WORD:process_alias}\] \[%{WORD:process_alias}\]\@\[%{WORD:process_alias}\]%{SPACE}LOG:%{SPACE}%{GREEDYDATA:content}"
                }
            }
        }
        else if [event][module] == "node" {
            grok {
                match => {
                    "message" => [
                    'time=%{TIMESTAMP_ISO8601:logdatetime} level=%{WORD:loglevel} msg=\"%{GREEDYDATA:content}\"',
                    'ip=::ffff:%{IPORHOST:ip} time=%{TIMESTAMP_ISO8601:logdatetime} method=%{GREEDYDATA:content}'
                    ]
                }
            }
        }
        else if [event][module] == "flog_generated" {
            grok {
                keep_empty_captures => true
                match => {
                    "message" => "%{IP:ip} %{GREEDYDATA} \[%{DATA:logdatetime}\] %{GREEDYDATA:content}"
                }
            }

        }
        else {
            grok {
                keep_empty_captures => true
                patterns_dir => ["/patterns"]
                match => {
                    "message" => [
                    '{"Timestamp":"%{DATESTAMP_EU:logdatetime}","EventId":%{NONNEGINT:event_id},"LogLevel":"%{WORD:loglevel}","Category":"%{DATA:category}","Message":%{GREEDYDATA:content}"',
                    "\[pid: %{NONNEGINT:pid}\|app: %{DATA:app}\|req: %{DATA:req}\] %{IPORHOST:ip} %{GREEDYDATA} \[%{HTTPDERROR_DATE:logdatetime}\] %{GREEDYDATA:content}",
                    "%{TIMESTAMP_ISO8601:logdatetime} %{MONGO3_SEVERITY:severity} %{MONGO3_COMPONENT:component}%{SPACE}(?:\[%{DATA:context}\])? %{GREEDYDATA:content}",
                    "%{DATETIME_YEAR_FIRST:logdatetime} \[%{GREEDYDATA:process_id}\] - %{LOGLEVEL:loglevel}%{SPACE}\[%{GREEDYDATA:urn}\] - %{GREEDYDATA:content}",
                    'time="?%{TIMESTAMP_ISO8601:logdatetime}"? level="?%{DATA:loglevel}"? msg="%{DATA:content}"(?:\s+%{GREEDYDATA})?',
                    '{"levelname": "%{WORD:loglevel}", "asctime": "%{TIMESTAMP_ISO8601:logdatetime}", "module": "%{DATA:module}", "message": "%{GREEDYDATA:content}',
                    '{"time":"%{NONNEGINT:timestamp_ms}", "source":"%{DATA:source"}, "status":"%{WORD:loglevel}", "message":"%{GREEDYDATA:content}"}',
                    "\[%{DATETIME_YEAR_FIRST:logdatetime}\] %{LOGLEVEL:loglevel} (\[%{DATA:logger}\]%{SPACE})?%{GREEDYDATA:content}",
                    "%{TIMESTAMP_ISO8601:logdatetime} %{NUMBER:thread_id} \[%{WORD:loglevel}\] %{GREEDYDATA:content}",
                    'time=\"%{TIMESTAMP_ISO8601:logdatetime}\" level=%{WORD:loglevel} msg=\"%{GREEDYDATA:content}\"',
                    '%{WORD:loglevel}:uvicorn:\(\'%{IPORHOST:ip_address}\', %{NONNEGINT:port}\) - %{GREEDYDATA:content}',
                    'level=%{WORD:loglevel} time=%{TIMESTAMP_ISO8601:logdatetime} msg=%{GREEDYDATA:content}',
                    '%{TIMESTAMP_ISO8601:logdatetime} - %{DATA:loglevel} - %{GREEDYDATA:content}',
                    "%{DATESTAMP:logdatetime} :: \[%{WORD:process_alias}\] %{GREEDYDATA:content}",
                    "%{IP:ip} %{GREEDYDATA} \[%{DATA:logdatetime}\] %{GREEDYDATA:content}"
                    ]
                }
            }
        }

        if "_grokparsefailure" not in [tags] {
            mutate {
                add_field => {
                    "[@metadata][was_parsed]" => "true"
                }

            }
        }

        # Docker:
        # %{TIMESTAMP_ISO8601:logdatetime} %{LOGLEVEL:loglevel} %{NONNEGINT:severity} --- \[%{SPACE}%{DATA:class}\] %{DATA:container}%{SPACE}: %{GREEDYDATA:content}

        if [logdatetime] {
            date {
                match => ["logdatetime", "MMM  d HH:mm:ss", "yyyy/MM/dd HH:mm:ss.SSS", "MMM d HH:mm:ss", "dd/MMM/yyyy:HH:mm:ss Z", "yyyy/MM/dd HH:mm:ss,SSS", "UNIX", "UNIX_MS", "MMM dd yyyy HH:mm:ss", "MMM d yyyy HH:mm:ss", "EEE MMM dd HH:mm:ss yyyy", "dd-MM-yyyy HH:mm:ss", "dd/MMM/yyyy HH:mm:ss", "ISO8601"]
                target => ["@timestamp"]
                remove_field => [ "logdatetime" ]
            }
        }
        ruby {
            code => "event.set('timestamp_unix', event.get('@timestamp').to_f)"
        }
    }
    if [@metadata][original] == "true" {
        mutate {
            remove_field => [
            "[agent][hostname]",
            "[agent][id]",
            "[agent][name]",
            "[agent][type]",
            "[agent][version]",
            "[agent][ephemeral_id]",
            "[host][architecture]",
            "[host][containerized]",
            "[host][id]",
            "[host][ip]",
            "[host][mac]",
            "[host][os]",
            "[ecs][version]",
            "[cloud][availability_zone]",
            "[cloud][instance][id]",
            "[cloud][instance][name]",
            "[cloud][machine][type]",
            "[cloud][provider]"
            ]
        }
    }
}
